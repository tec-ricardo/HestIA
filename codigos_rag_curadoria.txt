package br.com.hestia.rag.service;

    @Transactional
    public void indexarResposta(Long empresaId, Long respostaId) {
        Usuario usuario = usuarioAutorizado(empresaId);

        RespostaReutilizavel resposta = repository.findById(respostaId)
                .orElseThrow(() -> new IllegalArgumentException(
                        "Resposta nÃ£o encontrada."
                ));

        if (!empresaId.equals(resposta.getEmpresaId())) {
            throw new IllegalArgumentException(
                    "Resposta pertence a outra empresa."
            );
        }

        if (!resposta.getPerfisPermitidos().contains(usuario.getPerfil())) {
            throw new IllegalArgumentException(
                    "Perfil nÃ£o autorizado para indexaÃ§Ã£o."
            );
        }

        if (!resposta.isElegivel() || !Boolean.TRUE.equals(resposta.getReutilizavel())
                || !Boolean.FALSE.equals(resposta.getPossuiDadosSensiveis())) {
            throw new IllegalArgumentException(
                    "Resposta nÃ£o autorizada para indexaÃ§Ã£o."
            );
        }

        String promptIndexado = resposta.getPromptOriginal();
        String respostaIndexada = resposta.getRespostaGerada();
        String perfilAutorizado = usuario.getPerfil().name();
        String vetor = gerarVetor(promptIndexado);

        int atualizadas = jdbcTemplate.update(
                """
                UPDATE respostas_reutilizaveis
                SET embedding = ?,
                    embedding_vector = CAST(? AS vector)
                WHERE id = ?
                  AND empresa_id = ?
                  AND prompt_original = ?
                  AND resposta_gerada = ?
                  AND reutilizavel = true
                  AND status_qualidade = 'APROVADA'
                  AND (expira_em IS NULL OR expira_em > ?)
                  AND possui_dados_sensiveis = false
                  AND EXISTS (
                      SELECT 1
                      FROM resposta_perfis p
                      WHERE p.resposta_id = respostas_reutilizaveis.id
                        AND p.perfil = ?
                  )
                """,
                vetor,
                vetor,
                respostaId,
                empresaId,
                promptIndexado,
                respostaIndexada,
                java.time.LocalDateTime.now(),
                perfilAutorizado
        );

        if (atualizadas != 1) {
            throw new ResponseStatusException(
                    HttpStatus.CONFLICT,
                    "O conteÃºdo ou suas permissÃµes mudaram durante a indexaÃ§Ã£o. "
                            + "Execute a indexaÃ§Ã£o novamente."
            );
        }
    }

    @Transactional(readOnly = true)
    public List<RespostaReutilizavel> buscarSemelhantes(
            Long empresaId,
            String pergunta,
            int limite,
            double similaridadeMinima
    ) {
        Usuario usuario = usuarioAutorizado(empresaId);

        if (pergunta == null || pergunta.isBlank()) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Informe uma pergunta vÃ¡lida."
            );
        }

        if (limite < 1 || limite > 20) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "O limite deve estar entre 1 e 20."
            );
        }

        if (!Double.isFinite(similaridadeMinima)
                || similaridadeMinima < 0
                || similaridadeMinima > 1) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "A similaridade mÃ­nima deve estar entre zero e um."
            );
        }

        // NÃ£o chamar a OpenAI se nÃ£o existe nenhum vetor autorizado para pesquisar.
        // Sem fontes, o serviÃ§o deve retornar lista vazia, nÃ£o erro 503.
        Long quantidade = jdbcTemplate.queryForObject(
                """
                SELECT count(*) FROM respostas_reutilizaveis r
                WHERE r.empresa_id = ?
                  AND r.reutilizavel = true
                  AND r.status_qualidade = 'APROVADA'
                  AND (r.expira_em IS NULL OR r.expira_em > ?)
                  AND r.possui_dados_sensiveis = false
                  AND r.embedding_vector IS NOT NULL
                  AND EXISTS (
                      SELECT 1 FROM resposta_perfis p
                      WHERE p.resposta_id = r.id AND p.perfil = ?
                  )
                """,
                Long.class,
                empresaId,
                java.time.LocalDateTime.now(),
                usuario.getPerfil().name()
        );
        if (quantidade == null || quantidade == 0) {
            if (decisoes != null) decisoes.registrar(usuario, null,
                    "RECUPERACAO_VETORIAL", true, "SEM_CONTEUDO_AUTORIZADO");
            return List.of();
        }

        String vetor = gerarVetor(pergunta);

        List<Long> ids = jdbcTemplate.queryForList(
                """
                SELECT id
                FROM respostas_reutilizaveis
                WHERE empresa_id = ?
                  AND reutilizavel = true
                  AND status_qualidade = 'APROVADA'
                  AND (expira_em IS NULL OR expira_em > ?)
                  AND possui_dados_sensiveis = false
                  AND embedding_vector IS NOT NULL
                  AND EXISTS (
                      SELECT 1
                      FROM resposta_perfis p
                      WHERE p.resposta_id = respostas_reutilizaveis.id
                        AND p.perfil = ?
                  )
                  AND (embedding_vector <=> CAST(? AS vector)) <= ?
                ORDER BY embedding_vector <=> CAST(? AS vector)
                LIMIT ?
                """,
                Long.class,
                empresaId,
                java.time.LocalDateTime.now(),
                usuario.getPerfil().name(),
                vetor,
                1.0 - similaridadeMinima,
                vetor,
                limite
        );

        if (ids.isEmpty()) {
            if(decisoes!=null)decisoes.registrar(usuario,null,"RECUPERACAO_VETORIAL",true,"SEM_CONTEUDO_AUTORIZADO");
            return List.of();
        }

        Map<Long, RespostaReutilizavel> respostas =
                repository.findAllById(ids)
                        .stream()
                        .filter(r -> empresaId.equals(r.getEmpresaId()))
                        .filter(r -> r.getPerfisPermitidos()
                                .contains(usuario.getPerfil()))
                        .filter(r -> Boolean.TRUE.equals(r.getReutilizavel()))
                        .filter(RespostaReutilizavel::isElegivel)
                        .filter(r -> Boolean.FALSE.equals(
                                r.getPossuiDadosSensiveis()
                        ))
                        .collect(Collectors.toMap(
                                RespostaReutilizavel::getId,
                                Function.identity()
                        ));

        if(decisoes!=null)decisoes.registrar(usuario,null,"RECUPERACAO_VETORIAL",true,"FILTROS_ANTES_DO_LIMITE");
        return ids.stream()
                .map(respostas::get)
                .filter(Objects::nonNull)
                .toList();
    }

    private Usuario usuarioAutorizado(Long empresaId) {
        empresaAtual.validar(empresaId);

        Usuario usuario = usuarios
                .findById(empresaAtual.obterContexto().usuarioId())
                .orElseThrow(() -> new IllegalArgumentException(
                        "UsuÃ¡rio autenticado inexistente."
                ));

        acesso.validarEmpresa(usuario, empresaId);
        return usuario;
    }

    private String gerarVetor(String texto) {
        if (texto == null || texto.isBlank()) {
            throw new IllegalArgumentException(
                    "O texto nÃ£o pode estar vazio."
            );
        }

        List<Float> embedding = embeddingService.gerarEmbedding(texto);

        if (embedding == null || embedding.size() != DIMENSAO) {
            throw new IllegalStateException(
                    "DimensÃ£o do embedding invÃ¡lida."
            );
        }

        StringBuilder vetor = new StringBuilder("[");

        for (int i = 0; i < embedding.size(); i++) {
            Float valor = embedding.get(i);

            if (valor == null || !Float.isFinite(valor)) {
                throw new IllegalStateException(
                        "Embedding contÃ©m valor invÃ¡lido."
                );
            }

            if (i > 0) {
                vetor.append(",");
            }

            vetor.append(valor);
        }

        return vetor.append("]").toString();
    }

package br.com.hestia.rag.service;

import com.openai.client.OpenAIClient;
import com.openai.client.okhttp.OpenAIOkHttpClient;
import com.openai.models.embeddings.EmbeddingCreateParams;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class EmbeddingService {

    private static final String MODELO = "text-embedding-3-small";

    public List<Float> gerarEmbedding(String texto) {

        if (texto == null || texto.isBlank()) {
            throw new IllegalArgumentException("O texto nÃ£o pode estar vazio.");
        }

        OpenAIClient client = OpenAIOkHttpClient.fromEnv();

        var resposta = client.embeddings()
                .create(
                        EmbeddingCreateParams.builder()
                                .model(MODELO)
                                .input(texto)
                                .build()
                );

        return resposta.data()
                .get(0)
                .embedding();

package br.com.hestia.rag.service;

import br.com.hestia.reutilizavel.service.AcessoRespostaService;

import org.springframework.web.server.ResponseStatusException;

import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;

import static br.com.hestia.rag.dto.ResultadoReindexacaoDTO.Status.*;

@Service
public class ReindexacaoService {

    private final BuscaVetorialService busca;
    private final AcessoRespostaService acesso;

    public ReindexacaoService(
            BuscaVetorialService busca,
            AcessoRespostaService acesso
    ) {
        this.busca = busca;
        this.acesso = acesso;
    }

    public List<ResultadoReindexacaoDTO> reindexar(
            Usuario usuario,
            List<Long> respostas
    ) {
        Long empresaId = acesso.validarEmpresa(usuario, null);

        if (respostas == null || respostas.isEmpty()
                || respostas.size() > 20
                || respostas.stream().anyMatch(id -> id == null || id <= 0)) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Informe entre 1 e 20 identificadores positivos."
            );
        }

        List<ResultadoReindexacaoDTO> resultados = new ArrayList<>();

        for (Long id : new LinkedHashSet<>(respostas)) {
            try {
                busca.indexarResposta(empresaId, id);

                resultados.add(new ResultadoReindexacaoDTO(id, INDEXADA));
            } catch (ResponseStatusException exception) {
                int codigo = exception.getStatusCode().value();

                if (codigo == 401 || codigo == 403) {
                    throw exception;
                }

                resultados.add(new ResultadoReindexacaoDTO(
                        id,
                        codigo == 409 ? CONFLITO
                                : codigo == 404 ? INDISPONIVEL : FALHA
                ));
            } catch (IllegalArgumentException exception) {
                resultados.add(
                        new ResultadoReindexacaoDTO(id, INDISPONIVEL)
                );
            } catch (RuntimeException exception) {
                resultados.add(new ResultadoReindexacaoDTO(id, FALHA));
            }
        }

        return List.copyOf(resultados);
    }
}

import org.springframework.data.domain.Pageable;
import br.com.hestia.usuario.model.PerfilUsuario;

import java.util.List;

@Repository
public interface RespostaReutilizavelRepository
        extends JpaRepository<RespostaReutilizavel, Long> {

    @Query("""
            select r from RespostaReutilizavel r
            where r.empresaId = :empresaId
              and :perfil member of r.perfisPermitidos
              and r.reutilizavel = true and r.possuiDadosSensiveis = false
              and r.statusQualidade = br.com.hestia.reutilizavel.model.StatusQualidadeResposta.APROVADA
              and (r.expiraEm is null or r.expiraEm > :agora)
              and (:id is null or r.id = :id)
              and (cast(:prompt as string) is null or lower(r.promptOriginal) like
                   lower(concat('%', cast(:prompt as string), '%')) escape '!')
              and (:exigirEmbedding = false or (r.embedding is not null and trim(r.embedding) <> ''))
            order by r.dataCriacao desc, r.id desc
            """)
    List<RespostaReutilizavel> recuperarAutorizadas(
            @Param("empresaId") Long empresaId, @Param("perfil") PerfilUsuario perfil,
            @Param("id") Long id, @Param("prompt") String prompt,
            @Param("exigirEmbedding") boolean exigirEmbedding,@Param("agora") java.time.LocalDateTime agora, Pageable pageable);
